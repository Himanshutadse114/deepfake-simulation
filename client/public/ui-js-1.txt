function startWhatsAppSimulation() {
  go('voiceExperience');

function showWaTyping(show) {
  const chatBody = document.getElementById('waChatBody');
  const existing = document.getElementById('waTypingBubble');
  if (existing) existing.remove();
  const status = document.getElementById('waStatus');
  if(status) status.textContent = show ? 'typing...' : 'online';
  
  if (show) {
    const bubble = document.createElement('div');
    bubble.className = 'wa-bubble in';
    bubble.id = 'waTypingBubble';
    bubble.innerHTML = '<div class="wa-typing-dots"><span></span><span></span><span></span></div>';
    chatBody.appendChild(bubble);
    chatBody.scrollTop = chatBody.scrollHeight;
  }
}

function appendWaBubble(text, direction) {
  const chatBody = document.getElementById('waChatBody');
  const bubble = document.createElement('div');
  bubble.className = `wa-bubble ${direction}`;
  const now = new Date();
  const timeStr = `${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}`;
  bubble.innerHTML = `${text}<span class="wa-time">${timeStr} ${direction==='out'?'✓✓':''}</span>`;
  chatBody.appendChild(bubble);
  chatBody.scrollTop = chatBody.scrollHeight;
}

}
